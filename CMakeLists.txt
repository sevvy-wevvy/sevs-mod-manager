cmake_minimum_required(VERSION 3.10)
project(smm C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
file(GLOB SMM_SRC src/*.c src/ui/*.c src/core/*.c)
add_executable(smm ${SMM_SRC})
if(WIN32)
  target_link_libraries(smm ws2_32 gdi32 ole32 shell32 advapi32)
elseif(APPLE)
  target_link_libraries(smm "-framework Cocoa" "-framework CoreGraphics" pthread m)
else()
  target_link_libraries(smm m dl pthread)
endif()
